public with sharing class EmployeeController {

    @AuraEnabled
    public static String saveEmployee(

        String name,
        String email,
        Decimal salary,
        String department,
        Date joiningDate
    ) {

        try {

            // Duplicate Email Check

            List<Employee__c> existing = [

                SELECT Id
                FROM Employee__c
                WHERE Email__c = :email
                LIMIT 1
            ];

            if(!existing.isEmpty()){

                return 'EMAIL_EXISTS';
            }

            // Create Employee Record

            Employee__c emp =
                new Employee__c();

            emp.Name = name;

            emp.Email__c = email;

            emp.salary__c = salary;

            emp.department__c = department;

            emp.Join_Date__c = joiningDate;

            insert emp;

            return 'SUCCESS';
        }

        catch(Exception e){

            return e.getMessage();
        }
    }

    // FETCH RECORDS

    @AuraEnabled(cacheable=true)

    public static List<Employee__c> getEmployees(){

        return [

            SELECT
                Name,
                Email__c,
                salary__c,
                department__c,
                Join_Date__c

            FROM Employee__c
        ];
    }
}


EmployeeAuraController.js

({
    doInit : function(component, event, helper) {

        let action =
            component.get("c.getEmployees");

        action.setCallback(this, function(response){

            if(
                response.getState() === "SUCCESS"
            ){

                component.set(
                    "v.employees",
                    response.getReturnValue()
                );
            }
        });

        $A.enqueueAction(action);
    },

    handleSubmit : function(component, event, helper) {

        let name =
            component.get("v.employeeName");

        let email =
            component.get("v.email");

        let salary =
            component.get("v.salary");

        let department =
            component.get("v.department");

        let joiningDate =
            component.get("v.joiningDate");

        // VALIDATION

        if(
            !name ||
            name.trim().length < 3
        ){

            alert(
                "Employee Name must contain at least 3 characters"
            );

            return;
        }

        let emailRegex =
        /^[^\\s@]+@[^\\s@]+\\.[^\\s@]+$/;

        if(
            !emailRegex.test(email)
        ){

            alert("Invalid Email");

            return;
        }

        if(
            salary < 10000 ||
            salary > 500000
        ){

            alert(
                "Salary must be between 10000 and 500000"
            );

            return;
        }

        if(!department){

            alert(
                "Please select department"
            );

            return;
        }

        let today = new Date();

        let selectedDate =
            new Date(joiningDate);

        if(selectedDate > today){

            alert(
                "Joining date cannot be future date"
            );

            return;
        }

        // APEX CALL

        let action =
            component.get("c.saveEmployee");

        action.setParams({

            name : name,

            email : email,

            salary : salary,

            department : department,

            joiningDate : joiningDate
        });

        action.setCallback(this, function(response){

            let result =
                response.getReturnValue();

            if(result === "SUCCESS"){

                alert(
                    "Employee Saved Successfully"
                );

                $A.get(
                    'e.force:refreshView'
                ).fire();
            }

            else if(
                result === "EMAIL_EXISTS"
            ){

                alert(
                    "Email already exists"
                );
            }

            else{

                alert(result);
            }
        });

        $A.enqueueAction(action);
    }
})

public class email1 {
 
	public String Address {get;set;}
	public String subject {get;set;}
	public String body {get;set;}
 
	public PageReference sendEmail(){
 
    	// Email Validation
 
    	if(String.isBlank(Address) ||
           !Address.contains('@')){
 
        	ApexPages.addMessage(
 
            	new ApexPages.Message(
                    ApexPages.Severity.ERROR,
                	'Invalid Email ID'
            	)
        	);
 
        	return null;
    	}
 
    	// Create Email Message
 
        Messaging.SingleEmailMessage message =
        	new Messaging.SingleEmailMessage();
 
    	message.setToAddresses(
        	new String[] {Address}
    	);
 
        message.setSubject(subject);
 
        message.setPlainTextBody(body);
 
    	// Attachment
 
        Messaging.EmailFileAttachment attachment =
        	new Messaging.EmailFileAttachment();
 
        attachment.setFileName('Sample.txt');
 
    	attachment.setBody(
        	Blob.valueOf('Sample value for sending')
    	);
 
        attachment.setContentType('text/plain');
 
        message.setFileAttachments(
        	new Messaging.EmailFileAttachment[] {attachment}
    	);
 
    	// Send Email
 
        Messaging.SendEmailResult[] result =
 
        	Messaging.sendEmail(
            	new Messaging.SingleEmailMessage[] {message}
        	);
 
    	// Result Message
 
    	if(result[0].isSuccess()){
 
        	ApexPages.addMessage(
 
            	new ApexPages.Message(
                    ApexPages.Severity.CONFIRM,
                	'EMAIL SENT SUCCESSFULLY'
            	)
        	);
    	}
    	else{
 
        	ApexPages.addMessage(
 
            	new ApexPages.Message(
                    ApexPages.Severity.ERROR,
                    result[0].getErrors()[0].getMessage()
            	)
        	);
    	}
 
    	return null;
	}
}

public class StudentController {
 
	public Student__c student {get; set;}
 
	public List<Student__c> studentList {get; set;}
 
	// Constructor
	public StudentController(){
 
    	student = new Student__c();
 
    	loadstudents();
	}
 
	// Add Student
	public void addstudent(){
 
    	// Null Validation
    	if(student.Name == null ||
       	student.Rollno__c == null ||
       	student.class__c == null ||
       	student.mobile__c == null){
 
        	ApexPages.addMessage(
            	new ApexPages.Message(
                    ApexPages.Severity.ERROR,
                	'Do not leave fields empty'
            	)
        	);
 
        	return;
    	}
 
    	// Name Validation (only alphabets)
        if(!Pattern.matches('[a-zA-Z ]+', student.Name)){
 
        	ApexPages.addMessage(
            	new ApexPages.Message(
                    ApexPages.Severity.ERROR,
                	'Name should contain only alphabets'
            	)
        	);
 
        	return;
    	}
 
    	// Mobile Validation (exactly 10 digits)
        if(!Pattern.matches('[0-9]{10}', student.mobile__c)){
 
        	ApexPages.addMessage(
            	new ApexPages.Message(
                    ApexPages.Severity.ERROR,
                	'Mobile number must contain exactly 10 digits'
            	)
        	);
 
        	return;
    	}
 
    	// Duplicate Roll Number Check
    	List<Student__c> existing = [
 
        	SELECT Id
        	FROM Student__c
        	WHERE Rollno__c = :student.Rollno__c
        	AND Id != :student.Id
    	];
 
    	if(existing.size() > 0){
 
        	ApexPages.addMessage(
            	new ApexPages.Message(
                    ApexPages.Severity.ERROR,
                	'Roll number already exists'
            	)
        	);
 
        	return;
    	}
 
    	// Insert Record
    	upsert student;
 
    	// Clear Form
    	student = new Student__c();
 
    	// Reload Table
    	loadstudents();
	}
 
	// Load Records
	public void loadstudents(){
 
    	studentList = [
 
        	SELECT Name,
               	Rollno__c,
               	class__c,
               	mobile__c
 
        	FROM Student__c
    	];
	}
